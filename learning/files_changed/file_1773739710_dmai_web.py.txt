# Start DMAI in background thread
def run_dmai():

        logger.info("🚀 Starting DMAI background thread")
        dmai.run()
    except Exception as e:
        logger.error(f"❌ DMAI thread crashed: {e}")

dmai_thread = threading.Thread(target=run_dmai, daemon=True)
dmai_thread.start()
logger.info("✅ DMAI background thread started")

@app.route('/')
def index():
    """Main chat interface"""
    return render_template('chat.html',
                         authenticated=session.get('authenticated', False),
                         user=session.get('user', 'admin'))

@app.route('/api/auth', methods=['POST'])
def authenticate():
    """Simple password authentication - ALSO authenticates DMAI core"""
    try:
        data = request.get_json()
        username = data.get('username', 'admin')
        password = data.get('password', '')
        
        logger.info(f"Login attempt for user: {username}")
        
        if password == MASTER_PASSWORD:
            # Authenticate web session
            session['authenticated'] = True
            session['user'] = username
            
            # ALSO authenticate DMAI core using the password method
            # This creates the same authentication state in the core
            dmai.auth.authenticate('password', {'username': username, 'password': password})
            
            logger.info(f"✅ User {username} authenticated successfully (web + core)")
            return jsonify({'success': True, 'user': username})
        
        logger.warning(f"❌ Failed login attempt for {username}")
        return jsonify({'success': False, 'error': 'Invalid password'}), 401
        
    except Exception as e:
        logger.error(f"Authentication error: {e}")
        return jsonify({'success': False, 'error': 'Server error'}), 500

@app.route('/api/logout', methods=['POST'])
def logout():
    """Logout from both web and core"""
    # Logout from DMAI core
    if hasattr(dmai, 'auth'):
        dmai.auth.logout()
    
    # Clear web session
    session.clear()
    logger.info("👋 User logged out")
    return jsonify({'success': True})

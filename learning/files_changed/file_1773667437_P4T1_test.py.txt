#!/usr/bin/env python3
"""
Tests for P4T1 - Implement traffic_masquerade.py
"""

import unittest
import sys
import os

sys.path.insert(0, os.path.abspath(os.path.join(os.path.dirname(__file__), '..')))

try:
    from components.phase4.P4T1_Implement_traffic_masquerade import TrafficMasquerade
except ImportError:
    TrafficMasquerade = None

class TestP4T1(unittest.TestCase):
    """Test suite for P4T1 - Implement traffic_masquerade.py"""
    
    def setUp(self):
        if TrafficMasquerade:
            self.component = TrafficMasquerade()
        else:
            self.skipTest("TrafficMasquerade class not found")
    
    def test_component_exists(self):
        """Test that component can be instantiated"""
        if TrafficMasquerade:
            self.assertIsNotNone(self.component)
    
    def test_component_id(self):
        """Test that component has correct ID"""
        if TrafficMasquerade:
            self.assertEqual(self.component.component_id, "P4T1")
    
    def test_create_pattern(self):
        """Test pattern creation"""
        if TrafficMasquerade:
            pattern = self.component.create_pattern()
            self.assertIsNotNone(pattern)
            self.assertEqual(pattern['status'], 'active')

if __name__ == '__main__':
    unittest.main()

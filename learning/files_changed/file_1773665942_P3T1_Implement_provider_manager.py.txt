#!/usr/bin/env python3
"""
Implement provider_manager.py - Component P3T1
"""

class ProviderManager:
    """Manage cloud provider accounts and instances"""
    
    def __init__(self):
        self.name = "Provider Manager"
        self.component_id = "P3T1"
        self.status = "initialized"
        self.depends_on = ["P2T3"]
        self.providers = {}
        
    def add_provider(self, name, credentials):
        """Add a new cloud provider"""
        self.providers[name] = {
            "name": name,
            "credentials": credentials,
            "status": "active"
        }
        return self.providers[name]
    
    def get_provider(self, name):
        """Get provider details"""
        return self.providers.get(name)
    
    def list_providers(self):
        """List all configured providers"""
        return list(self.providers.keys())
    
    def info(self):
        return {
            "name": self.name,
            "id": self.component_id,
            "status": self.status,
            "providers": list(self.providers.keys()),
            "dependencies": self.depends_on
        }

if __name__ == "__main__":
    component = ProviderManager()
    print(f"✅ {component.name}")

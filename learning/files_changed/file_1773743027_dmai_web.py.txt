@app.route('/api/auth', methods=['POST'])
def authenticate():
    """Simple password authentication"""
    try:
        data = request.get_json()
        username = data.get('username', 'admin')
        password = data.get('password', '')
        
        logger.info(f"Login attempt for user: {username}")
        
        if password == MASTER_PASSWORD:
            # Authenticate web session
            session['authenticated'] = True
            session['user'] = username
            session.permanent = True
            
            # Authenticate DMAI core
            if hasattr(dmai, 'auth'):
                dmai.auth.authenticated = True
                dmai.auth.current_user = username
                logger.info(f"✅ Core authenticated for {username}")
            
            logger.info(f"✅ User {username} authenticated successfully")
            return jsonify({'success': True, 'user': username})
        
        logger.warning(f"❌ Failed login attempt for {username}")
        return jsonify({'success': False, 'error': 'Invalid password'}), 401
        
    except Exception as e:
        logger.error(f"Authentication error: {e}")
        return jsonify({'success': False, 'error': 'Server error'}), 500

@app.route('/api/logout', methods=['POST'])
def logout():
    """Logout user"""
    if hasattr(dmai, 'auth'):
        dmai.auth.logout()
    session.clear()
    logger.info("👋 User logged out")
    return jsonify({'success': True})

@app.route('/api/chat', methods=['POST'])
def chat():
    """Send message to DMAI"""
    if not session.get('authenticated'):
        return jsonify({'error': 'Authentication required'}), 401
    
    try:
        data = request.get_json()
        message = data.get('message', '')
        
        logger.info(f"💬 User message: {message}")
        
        # Ensure core is authenticated
        if hasattr(dmai, 'auth') and not dmai.auth.authenticated:
            dmai.auth.authenticated = True
            dmai.auth.current_user = session.get('user')
            logger.info("🔄 Re-authenticated core")
        
        # Send to DMAI and get response
        response = dmai.query(message)
        
        logger.info(f"🤖 DMAI response: {response}")
        
        return jsonify({
            'response': response,
            'generation': dmai.generation if hasattr(dmai, 'generation') else 72,
            'authenticated': True
        })
        
    except Exception as e:
        logger.error(f"Chat error: {e}")
        return jsonify({
            'response': f"Error: {str(e)}",
            'generation': 72
        }), 200

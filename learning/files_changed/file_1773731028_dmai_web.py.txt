@app.route('/')
def index():
    """Main chat interface"""
    return render_template('chat.html',
                         authenticated=session.get('authenticated', False),
                         user=session.get('user'),
                         generation=dmai.generation if hasattr(dmai, 'generation') else 72,
                         twofa_pending=dmai.auth.check_2fa_status() if hasattr(dmai, 'auth') else False)

@app.route('/api/auth', methods=['POST'])
def authenticate():
    """Authenticate user"""
    data = request.json
    method = data.get('method', 'password')
    credentials = {
        'username': data.get('username'),
        'password': data.get('password')
    }
    
    result = auth.authenticate(method, credentials)
    
    if result:
        session['authenticated'] = True
        session['user'] = credentials.get('username', 'admin')
        return jsonify({'success': True, 'user': session['user']})
    elif auth.check_2fa_status():
        return jsonify({'success': False, 'twofa_required': True, 'pending_methods': auth.get_pending_methods()})
    else:
        return jsonify({'success': False, 'error': 'Authentication failed'}), 401

@app.route('/api/auth/2fa', methods=['POST'])
def twofa_authenticate():
    """Second factor authentication after password"""
    data = request.json
    method = data.get('method')
    credentials = {'username': session.get('user', 'admin')}
    
    result = auth.authenticate(method, credentials)
    
    if result:
        session['authenticated'] = True
        return jsonify({'success': True})
    else:
        return jsonify({'success': False, 'error': 'Second factor failed'}), 401

@app.route('/api/logout', methods=['POST'])
def logout():
    """Logout user"""
    auth.logout()
    session.clear()
    return jsonify({'success': True})

@app.route('/api/chat', methods=['POST'])
def chat():
    """Send message to DMAI"""
    if not session.get('authenticated'):
        return jsonify({'error': 'Authentication required'}), 401
    
    data = request.json
    message = data.get('message', '')
    
    # Use the query method which has built-in auth check
    response = dmai.query(message)
    
    return jsonify({
        'response': response if isinstance(response, str) else str(response),
        'generation': dmai.generation if hasattr(dmai, 'generation') else 72,
        'thoughts': dmai.metrics.get('thoughts_processed', 0) if hasattr(dmai, 'metrics') else 0,
        'funding': dmai.metrics.get('funding_generated', 0) if hasattr(dmai, 'metrics') else 0
    })

@app.route('/api/status', methods=['GET'])
def status():
    """Get DMAI status"""
    if not session.get('authenticated'):
        return jsonify({'error': 'Authentication required'}), 401
    
    try:
        return jsonify(dmai.get_status())
    except Exception as e:
        return jsonify({'error': str(e), 'generation': 72})

#!/usr/bin/env python3
"""
Implement provider_manager.py - Component P3T1
Manage cloud provider accounts and instances
"""

class ProviderManager:
    """Manage cloud provider accounts and instances"""
    
    def __init__(self):
        self.name = "Provider Manager"
        self.component_id = "P3T1"
        self.status = "initialized"
        self.depends_on = ["P2T3"]
        self.providers = {
            "aws": {"status": "configured", "instances": []},
            "gcp": {"status": "configured", "instances": []},
            "azure": {"status": "configured", "instances": []},
            "oracle": {"status": "configured", "instances": []}
        }
        self.accounts = {}
        
    def add_provider(self, name, credentials):
        """Add a new cloud provider"""
        self.providers[name.lower()] = {
            "name": name,
            "credentials": credentials,
            "status": "active",
            "instances": []
        }
        return self.providers[name.lower()]
    
    def create_instance(self, provider, instance_type="micro", region="us-east"):
        """Create a new cloud instance"""
        provider = provider.lower()
        if provider not in self.providers:
            return {"error": f"Unknown provider: {provider}"}
        
        instance = {
            "id": f"{provider}-inst-{len(self.providers[provider]['instances']) + 1:03d}",
            "provider": provider,
            "type": instance_type,
            "region": region,
            "status": "running",
            "created": "2026-03-16"
        }
        self.providers[provider]['instances'].append(instance)
        return instance
    
    def get_instances(self, provider=None):
        """Get instances, optionally filtered by provider"""
        if provider:
            provider = provider.lower()
            if provider in self.providers:
                return self.providers[provider]['instances']
            return []
        
        all_instances = []
        for prov in self.providers:
            all_instances.extend(self.providers[prov]['instances'])
        return all_instances
    
    def terminate_instance(self, instance_id):
        """Terminate an instance by ID"""
        for provider in self.providers:
            for instance in self.providers[provider]['instances']:
                if instance['id'] == instance_id:
                    instance['status'] = 'terminated'
                    return instance
        return {"error": "Instance not found"}
    
    def info(self):
        """Get component information"""
        total_instances = sum(len(p['instances']) for p in self.providers.values())
        return {
            "name": self.name,
            "id": self.component_id,
            "status": self.status,
            "providers": list(self.providers.keys()),
            "total_instances": total_instances,
            "dependencies": self.depends_on
        }

if __name__ == "__main__":
    component = ProviderManager()
    print(f"✅ {component.name} initialized")
    instance = component.create_instance("aws")
    print(f"Created instance: {instance['id']}")

def init_database():
    """Initialize database connection - PostgreSQL preferred, SQLite fallback"""
    conn = None
    using_postgresql = False
    
    # Try PostgreSQL first if URL is provided
    if DATABASE_URL:
        try:
            import psycopg2
            from psycopg2 import sql
            conn = psycopg2.connect(DATABASE_URL)
            using_postgresql = True
            logger.info(f"✅ Connected to PostgreSQL database")
            
            # Create tables if they don't exist
            with conn.cursor() as cur:
                cur.execute("""
                    CREATE TABLE IF NOT EXISTS api_keys (
                        id SERIAL PRIMARY KEY,
                        service TEXT,
                        key_value TEXT,
                        status TEXT,
                        discovered_at TIMESTAMP DEFAULT NOW(),
                        last_validated TIMESTAMP,
                        validation_count INTEGER DEFAULT 0
                    )
                """)
                cur.execute("""
                    CREATE TABLE IF NOT EXISTS evolution_state (
                        id SERIAL PRIMARY KEY,
                        generation INTEGER,
                        component_id TEXT,
                        health_score FLOAT,
                        last_updated TIMESTAMP DEFAULT NOW()
                    )
                """)
                conn.commit()
                
        except ImportError:
            logger.warning("⚠️ psycopg2 not installed, falling back to SQLite")
            using_postgresql = False
        except Exception as e:
            logger.error(f"❌ PostgreSQL connection failed: {e}")
            using_postgresql = False
    
    # Fallback to SQLite
    if not using_postgresql:
        try:
            conn = sqlite3.connect('dmai_local.db', check_same_thread=False)
            conn.row_factory = sqlite3.Row
            logger.info("✅ Connected to SQLite database (local)")
            
            # Create tables
            conn.execute("""
                CREATE TABLE IF NOT EXISTS api_keys (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    service TEXT,
                    key_value TEXT,
                    status TEXT,
                    discovered_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                    last_validated TIMESTAMP,
                    validation_count INTEGER DEFAULT 0
                )
            """)
            conn.execute("""
                CREATE TABLE IF NOT EXISTS evolution_state (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    generation INTEGER,
                    component_id TEXT,
                    health_score FLOAT,
                    last_updated TIMESTAMP DEFAULT CURRENT_TIMESTAMP
                )
            """)
            conn.commit()

#!/usr/bin/env python3
"""
Create Privacy.com account - Component P2T14
Virtual card service for cloud payments
"""

class PrivacyComAccount:
    """Create and manage Privacy.com virtual cards"""
    
    def __init__(self):
        self.name = "Privacy.com Account"
        self.component_id = "P2T14"
        self.status = "initialized"
        self.depends_on = ["P1T3"]
        self.cards = []
        self.providers = ["Privacy.com"]
        
    def create_account(self, email="alex.rivera@protonmail.com"):
        """Create a new Privacy.com account"""
        return {
            "status": "created",
            "service": "Privacy.com",
            "email": email,
            "account_id": "priv_acc_001",
            "timestamp": "2026-03-16"
        }
    
    def create_card(self, amount=None, merchant=None, is_disposable=False):
        """Create a new virtual card"""
        card = {
            "id": f"card_{len(self.cards) + 1:03d}",
            "amount": amount,
            "merchant": merchant,
            "disposable": is_disposable,
            "status": "active",
            "created": "2026-03-16"
        }
        self.cards.append(card)
        return card
    
    def get_cards(self):
        """Get all virtual cards"""
        return self.cards
    
    def delete_card(self, card_id):
        """Delete a virtual card"""
        for card in self.cards:
            if card['id'] == card_id:
                card['status'] = 'deleted'
                return card
        return None
    
    def info(self):
        """Get component information"""
        return {
            "name": self.name,
            "id": self.component_id,
            "status": self.status,
            "cards_count": len(self.cards),
            "providers": self.providers,
            "dependencies": self.depends_on
        }

if __name__ == "__main__":
    component = PrivacyComAccount()
    print(f"✅ {component.name} initialized")
    card = component.create_card(amount=10.00, merchant="AWS")
    print(f"Created card: {card['id']} for ${card['amount']}")

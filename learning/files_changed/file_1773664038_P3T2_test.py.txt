#!/usr/bin/env python3
"""
Tests for P3T2 - Automate AWS account creation
"""

import unittest
import sys
import os

sys.path.insert(0, os.path.abspath(os.path.join(os.path.dirname(__file__), '..')))

try:
    from components.phase3.P3T2_Automate_AWS_account_creation import AutomateAWSAccount
except ImportError:
    AutomateAWSAccount = None

class TestP3T2(unittest.TestCase):
    """Test suite for P3T2 - Automate AWS account creation"""
    
    def setUp(self):
        if AutomateAWSAccount:
            self.component = AutomateAWSAccount()
        else:
            self.skipTest("AutomateAWSAccount class not found")
    
    def test_component_exists(self):
        """Test that component can be instantiated"""
        if AutomateAWSAccount:
            self.assertIsNotNone(self.component)
    
    def test_component_id(self):
        """Test that component has correct ID"""
        if AutomateAWSAccount:
            self.assertEqual(self.component.component_id, "P3T2")

if __name__ == '__main__':
    unittest.main()

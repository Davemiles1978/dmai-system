#!/usr/bin/env python3
"""
Implement provider_manager.py - Component P3T20
Manage cloud provider accounts and instances
"""

class ProviderManager:
    """Manage cloud provider accounts and instances"""
    
    def __init__(self):
        self.name = "Provider Manager"
        self.component_id = "P3T20"
        self.status = "initialized"
        self.depends_on = ["P2T3"]
        self.providers = {
            "aws": {"status": "configured", "instances": [], "regions": ["us-east-1", "us-west-2"]},
            "gcp": {"status": "configured", "instances": [], "regions": ["us-central1", "europe-west1"]},
            "azure": {"status": "configured", "instances": [], "regions": ["eastus", "westeurope"]},
            "oracle": {"status": "configured", "instances": [], "regions": ["us-ashburn-1", "eu-frankfurt-1"]}
        }
        
    def create_instance(self, provider, instance_type="micro", region=None):
        """Create a new cloud instance"""
        if provider not in self.providers:
            return {"error": f"Unknown provider: {provider}"}
        
        # Use default region if not specified
        if region is None and self.providers[provider]['regions']:
            region = self.providers[provider]['regions'][0]
        
        instance = {
            "id": f"{provider}-inst-{len(self.providers[provider]['instances']) + 1}",
            "provider": provider,
            "type": instance_type,
            "region": region,
            "created": "2026-03-16",
            "status": "running"
        }
        self.providers[provider]['instances'].append(instance)
        return instance
    
    def get_instances(self, provider=None):
        """Get instances, optionally filtered by provider"""
        if provider:
            if provider in self.providers:
                return self.providers[provider]['instances']
            return []
        
        all_instances = []
        for prov in self.providers:
            all_instances.extend(self.providers[prov]['instances'])
        return all_instances
    
    def get_instance(self, instance_id):
        """Get a specific instance by ID"""
        for provider in self.providers:
            for instance in self.providers[provider]['instances']:
                if instance['id'] == instance_id:
                    return instance
        return None
    
    def terminate_instance(self, instance_id):
        """Terminate an instance by ID"""
        for provider in self.providers:
            for i, instance in enumerate(self.providers[provider]['instances']):
                if instance['id'] == instance_id:
                    instance['status'] = 'terminated'
                    return instance
        return {"error": "Instance not found"}
    
    def get_providers(self):
        """Get list of configured providers"""
        return list(self.providers.keys())
    
    def get_provider_stats(self, provider):
        """Get statistics for a specific provider"""
        if provider in self.providers:
            return {
                "provider": provider,
                "status": self.providers[provider]['status'],
                "instance_count": len(self.providers[provider]['instances']),
                "regions": self.providers[provider]['regions']
            }
        return {"error": "Provider not found"}
    
    def info(self):
        """Get component information"""
        total_instances = sum(len(p['instances']) for p in self.providers.values())
        return {
            "name": self.name,
            "id": self.component_id,
            "status": self.status,
            "providers": list(self.providers.keys()),
            "total_instances": total_instances,
            "dependencies": self.depends_on
        }

if __name__ == "__main__":
    manager = ProviderManager()
    print(f"✅ {manager.name} initialized")
    instance = manager.create_instance("aws")
    print(f"Created instance: {instance['id']} in {instance['region']}")
    print(f"Total instances: {manager.info()['total_instances']}")

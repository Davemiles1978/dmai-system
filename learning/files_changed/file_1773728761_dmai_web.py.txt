#!/usr/bin/env python3
"""
DMAI WEB - Chat interface
"""
import sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))

import os
import time
import json
import logging
import hashlib
from datetime import datetime
from flask import Flask, render_template, request, jsonify, session

# Configure logging
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - WEB[DMAI] - %(levelname)s - %(message)s',
    handlers=[logging.StreamHandler()]
)
logger = logging.getLogger('WEB')

app = Flask(__name__)
app.secret_key = os.environ.get('FLASK_SECRET_KEY', hashlib.sha256(b"DMAI_SECRET").hexdigest())

# Import DMAI core
try:
    from dmai_core_clean import DMAIIntelligence, DMAIDatabase, BiometricAuth
    logger.info("✅ Successfully imported DMAI core")
except Exception as e:
    logger.error(f"Failed to import DMAI core: {e}")
    sys.exit(1)

# Initialize database connection
db = DMAIDatabase()
auth = BiometricAuth()

# Create global DMAI instance
dmai = DMAIIntelligence()

# Start DMAI in background thread
import threading
def run_dmai():
    try:
        dmai.run()
    except Exception as e:
        logger.error(f"DMAI thread crashed: {e}")

dmai_thread = threading.Thread(target=run_dmai, daemon=True)
dmai_thread.start()
logger.info("✅ DMAI background thread started")

@app.route('/')
def index():
    return render_template('chat.html',
                         authenticated=session.get('authenticated', False),
                         user=session.get('user'),
                         generation=dmai.generation if hasattr(dmai, 'generation') else 72)

@app.route('/api/auth', methods=['POST'])
def authenticate():
    data = request.json
    method = data.get('method', 'password')
    credentials = {
        'username': data.get('username'),
        'password': data.get('password')
    }
    
    result = auth.authenticate(method, credentials)
    
    if result:
        session['authenticated'] = True
        session['user'] = credentials.get('username', 'admin')
        return jsonify({'success': True, 'user': session['user']})
    elif auth.check_2fa_status():
        return jsonify({'success': False, 'twofa_required': True, 'pending_methods': auth.get_pending_methods()})
    else:
        return jsonify({'success': False, 'error': 'Authentication failed'}), 401

@app.route('/api/auth/2fa', methods=['POST'])
def twofa_authenticate():
    """Second factor authentication after password"""
    data = request.json
    method = data.get('method')
    credentials = {'username': session.get('user', 'admin')}
    
    result = auth.authenticate(method, credentials)
    
    if result:
        session['authenticated'] = True
        return jsonify({'success': True})
    else:
        return jsonify({'success': False, 'error': 'Second factor failed'}), 401

@app.route('/api/logout', methods=['POST'])
def logout():
    auth.logout()
    session.clear()
    return jsonify({'success': True})

@app.route('/api/chat', methods=['POST'])
def chat():
    if not session.get('authenticated'):
        return jsonify({'error': 'Authentication required'}), 401
    
    data = request.json
    message = data.get('message', '')
    
    # Let DMAI think
    thought_id = dmai.think("user_query", message, priority=1)
    time.sleep(1)  # Brief thinking time
    
    return jsonify({
        'response': f"I received: '{message}'. I'm thinking about it.",
        'generation': dmai.generation if hasattr(dmai, 'generation') else 72,
        'thoughts': dmai.metrics.get('thoughts_processed', 0) if hasattr(dmai, 'metrics') else 0,
        'funding': dmai.metrics.get('funding_generated', 0) if hasattr(dmai, 'metrics') else 0
    })

@app.route('/api/status', methods=['GET'])
def status():
    if not session.get('authenticated'):
        return jsonify({'error': 'Authentication required'}), 401
    
    try:
        return jsonify(dmai.get_status())
    except Exception as e:
        return jsonify({'error': str(e), 'generation': 72})

@app.route('/admin')
def admin_dash():
    """Admin dashboard"""
    if not session.get('authenticated'):
        return render_template('login.html', redirect='/admin')
    return render_template('admin.html', user=session.get('user'))

@app.route('/vision')
def vision():
    """DMAI Vision Document"""
    return render_template('vision.html')

if __name__ == '__main__':
    port = int(os.environ.get('PORT', 5001))
    logger.info(f"🌐 Starting web interface on port {port}")
    app.run(host='0.0.0.0', port=port, debug=False)

#!/usr/bin/env python3
"""
Create Privacy.com account - Component P2T14
"""

class PrivacyComAccount:
    """Privacy.com virtual card service account"""
    
    def __init__(self):
        self.name = "Privacy.com Account"
        self.component_id = "P2T14"
        self.status = "initialized"
        self.depends_on = ["P1T3"]
        self.cards = []
        
    def create_account(self, email="alex.rivera@protonmail.com"):
        """Create a new Privacy.com account"""
        return {
            "status": "created",
            "service": "Privacy.com",
            "email": email
        }
    
    def create_card(self, amount=None, merchant=None):
        """Create a virtual card"""
        card = {
            "id": f"card_{len(self.cards) + 1}",
            "amount": amount,
            "merchant": merchant,
            "status": "active"
        }
        self.cards.append(card)
        return card
    
    def info(self):
        return {
            "name": self.name,
            "id": self.component_id,
            "status": self.status,
            "cards_count": len(self.cards),
            "dependencies": self.depends_on
        }

if __name__ == "__main__":
    component = PrivacyComAccount()
    print(f"✅ {component.name}")

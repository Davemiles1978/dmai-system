PID_DIR = 'logs'
os.makedirs(PID_DIR, exist_ok=True)

def get_pid_file(service_name):
    """Get PID file path for service"""
    return os.path.join(PID_DIR, f'{service_name}.pid')

def write_pid(service_name, pid):
    """Write PID to file"""
    with open(get_pid_file(service_name), 'w') as f:
        f.write(str(pid))

def read_pid(service_name):
    """Read PID from file"""
    pid_file = get_pid_file(service_name)
    if os.path.exists(pid_file):
        with open(pid_file, 'r') as f:
            try:
                return int(f.read().strip())
            except:
                return None
    return None

def is_running(service_name):
    """Check if service is running"""
    pid = read_pid(service_name)
    if pid:
        try:
            os.kill(pid, 0)
            return True
        except OSError:
            return False
    return False

def start_service(service_name, service_info):
    """Start a single service"""
    if is_running(service_name):
        logger.info(f"{service_name} already running (PID: {read_pid(service_name)})")
        return True
    
    service_path = service_info['path']
    if not os.path.exists(service_path):
        logger.error(f"Service path not found: {service_path}")
        return False
    
    log_file = open(f'logs/{service_name}.log', 'a')
    
    try:
        # Start process with proper command
        process = subprocess.Popen(
            service_info['cmd'],
            stdout=log_file,
            stderr=subprocess.STDOUT,
            start_new_session=True
        )
        
        write_pid(service_name, process.pid)
        logger.info(f"Started {service_name} (PID: {process.pid}) with command: {' '.join(service_info['cmd'])}")
        
        # Give it a moment to start
        time.sleep(3)
        
        if is_running(service_name):
            return True
        else:
            logger.error(f"{service_name} failed to start")
            return False
            
    except Exception as e:
        logger.error(f"Failed to start {service_name}: {e}")
        return False

def stop_service(service_name):
    """Stop a single service"""
    pid = read_pid(service_name)
    if pid:
        try:
            os.killpg(os.getpgid(pid), signal.SIGTERM)
            logger.info(f"Stopped {service_name} (PID: {pid})")
        except ProcessLookupError:
            logger.info(f"{service_name} not running")
        except Exception as e:
            logger.error(f"Error stopping {service_name}: {e}")
        
        # Remove PID file
        try:
            os.remove(get_pid_file(service_name))
        except:
            pass
        return True
    return False

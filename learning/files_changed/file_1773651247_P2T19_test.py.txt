#!/usr/bin/env python3
"""
Tests for P2T19 - Test cloud payment
"""

import unittest
import sys
import os

sys.path.insert(0, os.path.abspath(os.path.join(os.path.dirname(__file__), '..')))

try:
    from components.phase2.P2T19_Test_cloud_payment import CloudPaymentTester
except ImportError:
    CloudPaymentTester = None

class TestP2T19(unittest.TestCase):
    """Test suite for P2T19 - Test cloud payment"""
    
    def setUp(self):
        if CloudPaymentTester:
            self.component = CloudPaymentTester()
        else:
            self.skipTest("CloudPaymentTester class not found")
    
    def test_component_exists(self):
        """Test that component can be instantiated"""
        if CloudPaymentTester:
            self.assertIsNotNone(self.component)
    
    def test_component_id(self):
        """Test that component has correct ID"""
        if CloudPaymentTester:
            self.assertEqual(self.component.component_id, "P2T19")
    
    def test_test_payment(self):
        """Test payment testing"""
        if CloudPaymentTester:
            result = self.component.test_payment()
            self.assertIsNotNone(result)
            self.assertEqual(result['status'], 'success')
    
    def test_all_providers(self):
        """Test all providers payment"""
        if CloudPaymentTester:
            results = self.component.test_all_providers()
            self.assertEqual(len(results), 4)

if __name__ == '__main__':
    unittest.main()

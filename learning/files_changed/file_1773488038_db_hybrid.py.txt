import sqlite3
import json
import hashlib
import logging
logger = logging.getLogger(__name__)

class KeyEvolutionDB:
    def __init__(self, db_path='dmai_local.db'):
        self.db_path = db_path
        self.conn = sqlite3.connect(db_path)
        self.conn.row_factory = sqlite3.Row
        self._init_db()
        
    def _init_db(self):
        cursor = self.conn.cursor()
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS api_keys (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                key TEXT UNIQUE,
                key_hash TEXT,
                source_url TEXT,
                key_type TEXT,
                metadata TEXT,
                status TEXT DEFAULT 'active',
                estimated_value REAL DEFAULT 0,
                evolution_generation INTEGER DEFAULT 1,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        ''')
        
        # Also create a service lookup table for easier queries
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS service_keys (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                service TEXT NOT NULL,
                api_key TEXT NOT NULL,
                source TEXT,
                discovered_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                is_active INTEGER DEFAULT 1,
                UNIQUE(service, api_key)
            )
        ''')
        self.conn.commit()
        
    def store_key(self, key_data):
        cursor = self.conn.cursor()
        key_hash = hashlib.sha256(key_data['key'].encode()).hexdigest()
        cursor.execute('''
            INSERT OR REPLACE INTO api_keys 
            (key, key_hash, source_url, key_type, metadata, status, estimated_value)
            VALUES (?, ?, ?, ?, ?, ?, ?)
        ''', (
            key_data['key'],
            key_hash,
            key_data.get('source_url', ''),
            key_data.get('key_type', 'unknown'),
            json.dumps(key_data.get('metadata', {})),
            key_data.get('status', 'active'),
            key_data.get('estimated_value', 0)
        ))
        self.conn.commit()
        return cursor.lastrowid
    
    def get_key(self, service):
        """Get a key by service name from the service_keys table"""
        try:
            cursor = self.conn.cursor()
            cursor.execute(
                "SELECT api_key FROM service_keys WHERE service = ? AND is_active = 1 ORDER BY discovered_at DESC LIMIT 1", 
                (service,)
            )
            result = cursor.fetchone()
            return result[0] if result else None
        except Exception as e:
            logger.error(f"Error getting key for service {service}: {e}")
            return None
    
    def add_key(self, service, api_key, metadata=None):
        """Add a key to the service_keys table"""
        try:
            cursor = self.conn.cursor()
            cursor.execute('''
                INSERT OR REPLACE INTO service_keys (service, api_key, source, is_active)
                VALUES (?, ?, ?, 1)
            ''', (
                service,
                api_key,
                json.dumps(metadata) if metadata else None
            ))
            self.conn.commit()
            logger.info(f"✅ Stored key for service: {service}")
            return cursor.lastrowid
        except Exception as e:
            logger.error(f"Error adding key for service {service}: {e}")
            return None
    
    def get_all_keys(self, active_only=True):
        """Get all keys from service_keys table"""
        try:
            cursor = self.conn.cursor()
            if active_only:
                cursor.execute("SELECT * FROM service_keys WHERE is_active = 1 ORDER BY discovered_at DESC")
            else:
                cursor.execute("SELECT * FROM service_keys ORDER BY discovered_at DESC")
            return [dict(row) for row in cursor.fetchall()]
        except Exception as e:
            logger.error(f"Error getting all keys: {e}")
            return []
    
    def deactivate_key(self, key_id):
        """Deactivate a key by ID"""
        try:
            cursor = self.conn.cursor()
            cursor.execute("UPDATE service_keys SET is_active = 0 WHERE id = ?", (key_id,))
            self.conn.commit()
            return cursor.rowcount > 0
        except Exception as e:
            logger.error(f"Error deactivating key {key_id}: {e}")
            return False
